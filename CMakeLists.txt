cmake_minimum_required(VERSION 3.20)
project(LogsOnFire VERSION 0.1.0 LANGUAGES CXX)

option(LOGSONFIRE_STATIC_RUNTIME "Use the static MSVC runtime" ON)
if(MSVC AND LOGSONFIRE_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(SQLite3 3.24 REQUIRED)
add_executable(logsonfire src/main.cpp)
target_compile_features(logsonfire PRIVATE cxx_std_17)
set_target_properties(logsonfire PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(logsonfire PRIVATE LOGSONFIRE_VERSION="${PROJECT_VERSION}")
if(TARGET SQLite3::SQLite3)
  target_link_libraries(logsonfire PRIVATE SQLite3::SQLite3)
else()
  target_link_libraries(logsonfire PRIVATE SQLite::SQLite3)
endif()
if(MSVC)
  target_compile_options(logsonfire PRIVATE /W4 /utf-8)
else()
  target_compile_options(logsonfire PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS logsonfire RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES python/logsonfire.py python/logsonfire_savings.py DESTINATION ${CMAKE_INSTALL_DATADIR}/logsonfire)
install(PROGRAMS scripts/logsonfire-agent DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES scripts/logsonfire-agent.ps1 DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE README.md BUILDING.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/logsonfire)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_system.py $<TARGET_FILE:logsonfire>)
  add_test(NAME packaging COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_packaging.py)
  add_test(NAME savings COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_savings.py $<TARGET_FILE:logsonfire>)
  add_test(NAME formats COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_formats.py $<TARGET_FILE:logsonfire>)
endif()

set(CPACK_PACKAGE_NAME logsonfire)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Local log analysis and compact evidence for agents")
set(CPACK_PACKAGE_VENDOR LogsOnFire)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "LogsOnFire contributors")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_GENERATOR "TGZ")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
endif()
include(CPack)
