cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LOGSONFIRE_VERSION LIMIT_COUNT 1)
string(STRIP "${LOGSONFIRE_VERSION}" LOGSONFIRE_VERSION)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" LOGSONFIRE_BASE_VERSION "${LOGSONFIRE_VERSION}")
if(NOT LOGSONFIRE_BASE_VERSION)
  message(FATAL_ERROR "VERSION must start with MAJOR.MINOR.PATCH")
endif()
project(LogsOnFire VERSION ${LOGSONFIRE_BASE_VERSION} LANGUAGES CXX)

option(LOGSONFIRE_STATIC_RUNTIME "Use the static MSVC runtime" ON)
if(MSVC AND LOGSONFIRE_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(SQLite3 3.24 REQUIRED)
add_executable(logsonfire src/main.cpp)
target_compile_features(logsonfire PRIVATE cxx_std_17)
set_target_properties(logsonfire PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(logsonfire PRIVATE LOGSONFIRE_VERSION="${LOGSONFIRE_VERSION}")
if(TARGET SQLite3::SQLite3)
  target_link_libraries(logsonfire PRIVATE SQLite3::SQLite3)
else()
  target_link_libraries(logsonfire PRIVATE SQLite::SQLite3)
endif()
if(MSVC)
  target_compile_options(logsonfire PRIVATE /W4 /utf-8)
else()
  target_compile_options(logsonfire PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS logsonfire RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES python/logsonfire.py python/logsonfire_savings.py python/logsonfire_version.py VERSION DESTINATION ${CMAKE_INSTALL_DATADIR}/logsonfire)
install(PROGRAMS scripts/logsonfire-agent DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES scripts/logsonfire-agent.ps1 DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE README.md BUILDING.md changelog.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/logsonfire)
install(FILES AGENTS.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/logsonfire)
install(DIRECTORY docs examples DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/logsonfire PATTERN "__pycache__" EXCLUDE)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_system.py $<TARGET_FILE:logsonfire>)
  add_test(NAME packaging COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_packaging.py)
  add_test(NAME savings COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_savings.py $<TARGET_FILE:logsonfire>)
  add_test(NAME formats COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_formats.py $<TARGET_FILE:logsonfire>)
  add_test(NAME output COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_output.py $<TARGET_FILE:logsonfire>)
  add_test(NAME version COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_version.py $<TARGET_FILE:logsonfire>)
  set_tests_properties(integration packaging savings formats output version PROPERTIES ENVIRONMENT "LOGSONFIRE_OUTPUT=json")
endif()

set(CPACK_PACKAGE_NAME logsonfire)
set(CPACK_PACKAGE_VERSION ${LOGSONFIRE_VERSION})
if(LOGSONFIRE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+-([^+]+)")
  set(CPACK_DEBIAN_PACKAGE_VERSION "${PROJECT_VERSION}~${CMAKE_MATCH_1}")
  set(CPACK_RPM_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_RPM_PACKAGE_RELEASE "0.${CMAKE_MATCH_1}")
endif()
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Local log analysis and compact evidence for agents")
set(CPACK_PACKAGE_VENDOR LogsOnFire)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "LogsOnFire contributors")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_GENERATOR "TGZ")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
endif()
include(CPack)
